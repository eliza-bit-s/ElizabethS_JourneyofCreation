Interactive Fantasy Game

By: Elizabeth Suarez

Date: ----

Language: Python

After a trend on tiktok took over my For You Page, I decided to put my own twist on it and translate the idea to an interactive python game. The general idea of the trend is still there despite the new mode of ingesting the media: a small fairytale story where the users decisions result in more positive or more negative outcomes based on the points they gain throughout the journey they're guided through. For my implementation, I decided that I'd stick to a singular storyline with the user being able to choose how and to what degree they interacted with the characters and events in the story as opposed to exposing the user to several different storylines with distinct endings like other famous, more advanced interactive games are praised for.

def choice():
    choice = "p"
    while ( (choice != "A") and (choice != "B") and (choice != "C") ):
        choice = (str(input("Enter your choice: A, B, or C? ")))[:1]
        choice = choice.upper()
        if (( (choice != "A") and (choice != "B") and (choice != "C") ) == True):
            print("Invalid Answer. Please type one of the following: A, B, or C.")
    return choice

def find_altendings(endtype):
    not_endtype3= "D"

    nty3 = "Decent"

        bool = "((opt != 'W') and (opt != 'D'))"

        bool = "((opt != 'B') and (opt != 'W'))"

def opt_altendings(logic, specifics):
    opt1, opt2 = specifics[0], specifics[1]             #update these
    opt = "p"
    print(logic)
    while ( eval(logic) ): #conditional for user picking option failsafe
        opt = (str(input("Enter your choice: " + opt1 + " or " + opt2 + "? ")))   #ask for ending
        opt = opt.upper()[:1]
        print(opt)
        if ( (eval(logic)) == True):
            print("Invalid Answer. Please type one of the following: " + opt1 + " or " + opt2 + ".")
    return opt  #return ending type

def ask_showaltends():    
    choice1 = "p"       #question asking if want to see another alternate ending
    while ( (choice1 != "YES") and (choice1 != "NO")  ):
        choice1 = (str(input("Enter your choice: Yes or No? ")))[:3]
        choice1 = choice1.upper()
        if (( (choice1 != "YES") and (choice1 != "NO") ) == True):
            print("Invalid Answer. Please type one of the following: Yes or No.")
    return choice1

def show_altendings(endtype):
    print("\nThank you for playing! Would you like to see alternate endings? Reply Yes or No")
    choice1 = ask_showaltends()
    while (choice1 == "YES"):  #yes I want to see an alternate ending
        bool2, spec_type = find_altendings(endtype)  #which alternate ending
        ending_chosen = opt_altendings(bool2, spec_type) #user chooses ending
        if ending_chosen == "W":                #showing ending chosen
            print("Now showing: Worst Ending")
            show_worstending()
            print("Would you like to see one of two more alternate endings? Reply Yes or No.")
            choice1 = ask_showaltends()

        elif ending_chosen == "B":
            print("Now showing: Best Ending")
            show_bestending()
            print("Would you like to see one of two more alternate endings? Reply Yes or No.")
            choice1 = ask_showaltends()

        else:
            print("Now showing: Decent Ending")
            show_decentending()
            print("Would you like to see one of two more alternate endings? Reply Yes or No.")
            choice1 = ask_showaltends()

    else:
        print("You may now exit the game.")
